```python
def numIslands(grid:List[List[str]]) -> int:
	n, m = len(grid), len(grid[0])
	count = 0
	def bfs(i, j):
		queue = []
		queue.appand(i, j)
		grid[i][j] = "0"
		while queue:
			i, j = queue.leftpop()
			for x, y in [[1,0], [-1,0], [0,1], [0,-1]]:
				row, col = i+x, j+y
				if (0 <=row<n and 0<=col<m and grid[row][col]="1")
					grid[row][col] = "0"
					queue.append(row, col)
	for i in range(n):
		for j in range(m):
			if grid[i][j] = "1":
				count += 1
				bfs(i, j)
	return count
```
